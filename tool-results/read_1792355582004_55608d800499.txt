   188→// ─── خواندن صوتی (TTS فارسی — v10.1.2): اول API داشبورد، در نبودش صدای محلی مرورگر ───
   189→const DIGIT_WORDS = ['صفر', 'یک', 'دو', 'سه', 'چهار', 'پنج', 'شش', 'هفت', 'هشت', 'نه', 'ده', 'یازده', 'دوازده'];
   190→const speakCache = new Map();
   191→let speakingBusy = false;
   192→
   193→// fallback محلی: Web Speech API خود مرورگر (fa-IR) — در استقرار مستقل (گیت‌هاب Pages) هم کار می‌کند؛
   194→// اگر صدای فارسی نصب نباشد، موتور پیش‌فرض تلاش می‌کند و در بدترین حالت بی‌صدا می‌ماند.
   195→function localSpeak(text) {
   196→  return new Promise(resolve => {
   197→    try {
   198→      if (typeof window === 'undefined' || !('speechSynthesis' in window)) return resolve(false);
   199→      const synth = window.speechSynthesis;
   200→      try { synth.cancel(); } catch { /* بی‌اهمیت */ }
   201→      const u = new SpeechSynthesisUtterance(text);
   202→      u.lang = 'fa-IR';
   203→      u.rate = 0.9;
   204→      const voices = typeof synth.getVoices === 'function' ? (synth.getVoices() || []) : [];
   205→      const faVoice = voices.find(v => v && /^fa/i.test(v.lang || ''));
   206→      if (faVoice) u.voice = faVoice;
   207→      let settled = false;
   208→      const done = ok => { if (!settled) { settled = true; resolve(ok); } };
   209→      u.onend = () => done(true);
   210→      u.onerror = () => done(false);
   211→      synth.speak(u);
   212→      setTimeout(() => done(true), 20000); // محافظ: اگر رویداد end/error نیامد
   213→    } catch { resolve(false); }
   214→  });
   215→}

   217→// گویندهٔ مشترک: صدا از API داشبورد (اگر در دسترس بود) → وگرنه صدای محلی مرورگر → وگرنه بی‌صدا.
   218→// فقط URLهای موفق کش می‌شوند تا شکست API در تلاش بعدی دوباره امتحان شود.
   219→async function speakFa(text, btn = null) {
   220→  if (!text || speakingBusy) return;
   221→  speakingBusy = true;
   222→  const prevLabel = btn ? btn.textContent : null;
   223→  if (btn) btn.textContent = '... در حال خواندن';
   224→  try {
   225→    let url = speakCache.get(text) || null;
   226→    if (!url && typeof fetch === 'function') {
   227→      try {
   228→        const res = await fetch('/api/tts', {
   229→          method: 'POST',
   230→          headers: { 'Content-Type': 'application/json' },
   231→          body: JSON.stringify({ text, speed: 0.9 })
   232→        });
   233→        if (res.ok) {
   234→          url = URL.createObjectURL(await res.blob());
   235→          speakCache.set(text, url);
   236→        }
   237→      } catch { /* داشبورد در دسترس نیست — fallback محلی */ }
   238→    }
   239→    if (url) {
   240→      await new Audio(url).play();
   241→    } else {
   242→      const ok = await localSpeak(text);
   243→      if (!ok && btn && btn.id === 'question-speak') notice('خواندن صوتی در دسترس نیست.', true);
   244→    }
   245→  } catch { /* TTS اختیاری است — بی‌صدا */ }
   246→  finally {
   247→    speakingBusy = false;
   248→    if (btn && prevLabel) btn.textContent = prevLabel;
   249→  }
   250→}

   330→function buildProgressLine(systemsNow) {
   331→  if (!systemsNow?.length) return '';
   332→  const parts = systemsNow.map(s => {
   333→    const current = s.currentDepth || 0;
   334→    const nextDepth = Math.min(3, current + 1);
   335→    const layer = DEPTH_LAYERS[`${s.sectorId}-${nextDepth}`];
   336→    const need = nextDepth > current ? layer?.unlock?.minPercent : null;
   337→    const suffix = need ? ` (لایهٔ ${fa(nextDepth)} با ${fa(need)}٪)` : '';
   338→    return `${BUILDING_NAMES[s.sectorId] || 'بنا'}: ${fa(s.percent)}٪${suffix}`;
   339→  });
   340→  return `📍 پیشرفت ایستگاه — ${parts.join(' • ')}`;
   341→}

   591→async function handleAnswerSubmit(userAnswer) {
   592→  const session = state.session;
   593→  const question = session.questions[activeQuestionIndex];
   594→  const expected = question.f1 * question.f2;
   595→  const isCorrect = Number(normalizeDigits(userAnswer)) === expected;
   596→  // زمان‌سنجی سؤال (گرد تا یک دهم ثانیه) — برای receipt و تلاش‌نگار
   597→  const elapsedSec = questionStartedAt
   598→    ? Math.round(((Date.now() - questionStartedAt) / 1000) * 10) / 10
   599→    : 0;
   600→
   601→  const result = stationLayer.cells.onAnswer(isCorrect);
   602→
   603→  if (!isCorrect) {
   604→    // تلاش‌نگار فقط‌الحاقی (CEAP §4.1): پاسخ‌های غلط حذف نمی‌شوند — شاهد تشخیص گلوگاه
   605→    session.wrongAttempts = session.wrongAttempts || [];
   606→    session.wrongAttempts.push({
   607→      questionId: question.questionId,
   608→      submitted: Number(normalizeDigits(userAnswer)),
   609→      timeSec: elapsedSec,
   610→      at: new Date().toISOString()
   611→    });
   612→    $('question-feedback').textContent = result.feedback;
   613→    void playStationAudio(result.sound);
   614→    // نردبان خودکار پلهٔ ۱ (v10.1.2): بعد از خطا، ملایم‌ترین راهنما («به عددها نگاه کن»)
   615→    // همان لحظه دیده و شنیده می‌شود — بدون افشای پاسخ (assertNoLeak پلهٔ ۱)
   616→    if (hintLevel === 0) {
   617→      hintLevel = 1;
   618→      session.hintsMax = Math.max(session.hintsMax || 0, hintLevel);
   619→      const autoBtn = $('question-guide-button');
   620→      if (autoBtn) autoBtn.classList.add('attract');
   621→      showGuideStepUp(question);
   622→    }
   623→    return;
   624→  }
   625→
   626→  void playStationAudio(result.sound);
   627→  $('question-feedback').textContent = '';
   628→
   629→  const eventId = 'evt_' + crypto.randomUUID();
   630→  session.answers = session.answers || {};
   631→  session.answers[question.questionId] = {
   632→    eventId,
   633→    questionId: question.questionId,
   634→    answer: expected,
   635→    submittedAt: new Date().toISOString()
   636→  };
   637→  session.receipts = session.receipts || {};
   638→  session.receipts[eventId] = {
   639→    eventId,
   640→    questionId: question.questionId,
   641→    correct: true,
   642→    status: 'ACKNOWLEDGED',
   643→    timeSec: elapsedSec, // زمان طی‌شده روی این سؤال (فیلد اضافی مجاز)
   644→    serverTime: new Date().toISOString()
   645→  };
   646→
   647→  if (result.action === 'next-question') {
   648→    activeQuestionIndex++;
   649→    showGuide = false;
   650→    await saveState(state);
   651→    renderQuestionView();
   652→  } else if (result.action === 'building-complete') {
   653→    session.status = 'FINALIZED';
   654→    session.score = session.questions.length;
   655→    session.total = session.questions.length;
   656→
   657→    state.stationRecords = keepStationRecord(state.stationRecords || [], session, {
   658→      status: 'FINALIZED',
   659→      score: session.score,
   660→      total: session.total
   661→    });
   662→
   663→    // تلاش‌نگار فقط‌الحاقی (CEAP §4.1): تاریخچهٔ کامل جلسه موازی با رکورد «بهترین» نگه داشته می‌شود
   664→    // keepStationRecord دست‌نخورده می‌ماند (نمایش «بهترین» UI است)؛ اینجا فقط ضمیمه می‌کنیم.
   665→    state.attemptLog = state.attemptLog || [];
   666→    const attemptEntry = {
   667→      eventId: 'evt_' + crypto.randomUUID(),
   668→      activityId: session.activityId,
   669→      sectorId: session.sectorId,
   670→      depth: session.depth,
   671→      missionId: session.missionId || null,
   672→      type: session.activityType || 'core',
   673→      total: session.total,
   674→      correct: session.questions.filter(q => session.answers?.[q.questionId]).length,
   675→      wrongCount: (session.wrongAttempts || []).length,
   676→      hintsMax: session.hintsMax || 0,
   677→      completedAt: new Date().toISOString()
   678→    };
   679→    state.attemptLog = state.attemptLog.slice(-499).concat(attemptEntry); // سقف ۵۰۰ ورودی
   680→
   681→    await saveState(state);
   682→
   683→    // پل داشبورد معلم: ارسال آتش‌وفراموش‌شدهٔ جلسه به سرور محلی (POST /api/sessions)
   684→    // اگر بازی مستقل از داشبورد اجرا شود یا شبکه در دسترس نباشد، خطاها بی‌صدا نادیده گرفته می‌شوند.
   685→    try {
   686→      if (typeof fetch === 'function') {
   687→        const dashProfile = (profiles || []).find(p => p.studentId === selectedId);
   688→        // سکتور می‌تواند رشتهٔ بنا ('station-power') یا شمارهٔ سکتور مأموریت جانبی (۱/۲/۳) باشد
   689→        const sectorKey = String(session.sectorId ?? '');
   690→        const sectorNames = { 'station-power': 'station-power', 'station-antenna': 'station-antenna', 'station-launch': 'station-launch', '1': 'station-power', '2': 'station-antenna', '3': 'station-launch' };
   691→        const sectorOffsets = { 'station-power': 0, 'station-antenna': 3, 'station-launch': 6, '1': 0, '2': 3, '3': 6 };
   692→        const buildingName = sectorNames[sectorKey] ?? 'unknown';
   693→        const sectorOffset = sectorOffsets[sectorKey] ?? 0;
   694→        const firstAt = Object.values(session.receipts || {})[0]?.serverTime;
   695→        const lastAt = Object.values(session.receipts || {}).slice(-1)[0]?.serverTime;
   696→        const spanSec = firstAt && lastAt ? Math.max(0, Math.round((new Date(lastAt) - new Date(firstAt)) / 1000)) : undefined;
   697→        // شناسهٔ قطعی رویداد برای همگام‌سازی idempotent — همان تلاش = همان eventId؛
   698→        // ریترای شبکه رکورد تکراری نمی‌سازد، اما بازی دوبارهٔ همان بنا رویداد تازه می‌سازد.
   699→        const syncEventId = ('evt_' + String(session.activityId ?? ('sync-' + sectorKey)) + '_' + String(firstAt ?? new Date().toISOString()))
   700→          .replace(/\s/g, '-')
   701→          .slice(0, 120);
   702→        void fetch('/api/sessions', {
   703→          method: 'POST',
   704→          headers: { 'Content-Type': 'application/json' },
   705→          keepalive: true,
   706→          body: JSON.stringify({
   707→            eventId: syncEventId,
   708→            studentName: dashProfile?.displayName || 'بازیکن ایستگاه',
   709→            buildingId: buildingName,
   710→            layer: Math.min(9, sectorOffset + (session.depth ?? 2)),
   711→            mode: session.activityType === 'side' ? 'side' : 'station',
   712→            missionId: session.missionId || undefined,
   713→            durationSec: spanSec,
   714→            hintsMax: session.hintsMax || 0, // بیشترین پلهٔ نردبان راهنما — تشخیص تکدی سرنخ
   715→            results: (session.questions || []).map(q => {
   716→              const answered = !!(session.answers && session.answers[q.questionId]);
   717→              // شواهد فقط‌الحاقی: آخرین پاسخ واقعی دانش‌آموز برای هر سؤال
   718→              const wrongs = (session.wrongAttempts || []).filter(w => w && w.questionId === q.questionId);
   719→              const lastWrong = wrongs.length ? wrongs[wrongs.length - 1] : null;
   720→              const receipt = Object.values(session.receipts || {}).find(r => r && r.questionId === q.questionId);
   721→              const item = {
   722→                a: q.f1 ?? 0,
   723→                b: q.f2 ?? 0,
   724→                correct: answered
   725→              };
   726→              if (answered) {
   727→                item.submitted = q.f1 * q.f2; // نهایی درست
   728→              } else if (lastWrong && Number.isFinite(lastWrong.submitted)) {
   729→                item.submitted = lastWrong.submitted; // آخرین پاسخ غلط (شاهد تشخیص)
   730→              }
   731→              if (receipt && Number.isFinite(receipt.timeSec)) {
   732→                item.timeSec = receipt.timeSec;
   733→              }
   734→              return item;
   735→            })
   736→          })
   737→        }).catch(() => {});
   738→      }
   739→    } catch { /* داشبورد اختیاری است */ }
   740→
   741→    currentView = 'map';
   742→    showGuide = false;
   743→    void playStationAudio(SOUND_EVENTS.pathLit);
   744→    void refreshTeacherMissions();
   745→    render();
   746→  }
   747→}

   749→function generateQuestionsForDepth(buildingId, depth) {

   751→  const layer = DEPTH_LAYERS[layerKey] || DEPTH_LAYERS['1-1'];
   752→  const factors = layer.focus?.factors || [buildingId + 1];
   753→  const count = layer.questionCount || 8;
   754→
   755→  const questions = [];
   756→  for (let i = 0; i < count; i++) {
   757→    const f1 = factors[i % factors.length];
   758→    const f2 = (i % 9) + 2;
   759→    questions.push({
   760→      questionId: `q_${buildingId}_${depth}_${i + 1}`,
   761→      f1,
   762→      f2
   763→    });
   764→  }
   765→  return questions;
   766→}
