   189→   189→const DIGIT_WORDS = ['صفر', 'یک', 'دو', 'سه', 'چهار', 'پنج', 'شش', 'هفت', 'هشت', 'نه', 'ده', 'یازده', 'دوازده'];

   194→   194→// اگر صدای فارسی نصب نباشد، موتور پیش‌فرض تلاش می‌کند و در بدترین حالت بی‌صدا می‌ماند.
   195→   195→function localSpeak(text) {
   196→   196→  return new Promise(resolve => {
   197→   197→    try {
   198→   198→      if (typeof window === 'undefined' || !('speechSynthesis' in window)) return resolve(false);
   199→   199→      const synth = window.speechSynthesis;
   200→   200→      try { synth.cancel(); } catch { /* بی‌اهمیت */ }
   201→   201→      const u = new SpeechSynthesisUtterance(text);
   202→   202→      u.lang = 'fa-IR';
   203→   203→      u.rate = 0.9;
   204→   204→      const voices = typeof synth.getVoices === 'function' ? (synth.getVoices() || []) : [];
   205→   205→      const faVoice = voices.find(v => v && /^fa/i.test(v.lang || ''));
   206→   206→      if (faVoice) u.voice = faVoice;
   207→   207→      let settled = false;
   208→   208→      const done = ok => { if (!settled) { settled = true; resolve(ok); } };
   209→   209→      u.onend = () => done(true);
   210→   210→      u.onerror = () => done(false);
   211→   211→      synth.speak(u);
   212→   212→      setTimeout(() => done(true), 20000); // محافظ: اگر رویداد end/error نیامد
   213→   213→    } catch { resolve(false); }
   214→   214→  });
   215→   215→}

   218→   218→// فقط URLهای موفق کش می‌شوند تا شکست API در تلاش بعدی دوباره امتحان شود.
   219→   219→async function speakFa(text, btn = null) {
   220→   220→  if (!text || speakingBusy) return;
   221→   221→  speakingBusy = true;
   222→   222→  const prevLabel = btn ? btn.textContent : null;
   223→   223→  if (btn) btn.textContent = '... در حال خواندن';
   224→   224→  try {
   225→   225→    let url = speakCache.get(text) || null;
   226→   226→    if (!url && typeof fetch === 'function') {
   227→   227→      try {
   228→   228→        const res = await fetch('/api/tts', {
   229→   229→          method: 'POST',
   230→   230→          headers: { 'Content-Type': 'application/json' },
   231→   231→          body: JSON.stringify({ text, speed: 0.9 })
   232→   232→        });
   233→   233→        if (res.ok) {
   234→   234→          url = URL.createObjectURL(await res.blob());
   235→   235→          speakCache.set(text, url);
   236→   236→        }
   237→   237→      } catch { /* داشبورد در دسترس نیست — fallback محلی */ }
   238→   238→    }
   239→   239→    if (url) {
   240→   240→      await new Audio(url).play();
   241→   241→    } else {
   242→   242→      const ok = await localSpeak(text);
   243→   243→      if (!ok && btn && btn.id === 'question-speak') notice('خواندن صوتی در دسترس نیست.', true);
   244→   244→    }
   245→   245→  } catch { /* TTS اختیاری است — بی‌صدا */ }
   246→   246→  finally {
   247→   247→    speakingBusy = false;
   248→   248→    if (btn && prevLabel) btn.textContent = prevLabel;
   249→   249→  }
   250→   250→}

   723→   723→                b: q.f2 ?? 0,
   724→   724→                correct: answered
   725→   725→              };
   726→   726→              if (answered) {
   727→   727→                item.submitted = q.f1 * q.f2; // نهایی درست
   728→   728→              } else if (lastWrong && Number.isFinite(lastWrong.submitted)) {
   729→   729→                item.submitted = lastWrong.submitted; // آخرین پاسخ غلط (شاهد تشخیص)
   730→   730→              }
   731→   731→              if (receipt && Number.isFinite(receipt.timeSec)) {
   732→   732→                item.timeSec = receipt.timeSec;
   733→   733→              }
   734→   734→              return item;
   735→   735→            })
   736→   736→          })
   737→   737→        }).catch(() => {});
   738→   738→      }
   739→   739→    } catch { /* داشبورد اختیاری است */ }
   740→   740→
   741→   741→    currentView = 'map';
   742→   742→    showGuide = false;
   743→   743→    void playStationAudio(SOUND_EVENTS.pathLit);
   744→   744→    void refreshTeacherMissions();
   745→   745→    render();
   746→   746→  }

   749→   749→function generateQuestionsForDepth(buildingId, depth) {
   750→   750→  const layerKey = `${buildingId}-${depth}`;
   751→   751→  const layer = DEPTH_LAYERS[layerKey] || DEPTH_LAYERS['1-1'];
   752→   752→  const factors = layer.focus?.factors || [buildingId + 1];

   755→   755→  const questions = [];
   756→   756→  for (let i = 0; i < count; i++) {
   757→   757→    const f1 = factors[i % factors.length];

   759→   759→    questions.push({
   760→   760→      questionId: `q_${buildingId}_${depth}_${i + 1}`,
   761→   761→      f1,
   762→   762→      f2
   763→   763→    });
   764→   764→  }
   765→   765→  return questions;
   766→   766→}
