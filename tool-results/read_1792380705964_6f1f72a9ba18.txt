   220→// ─── خواندن صوتی (TTS فارسی — v10.1.2): اول API داشبورد، در نبودش صدای محلی مرورگر ───
   221→const DIGIT_WORDS = ['صفر', 'یک', 'دو', 'سه', 'چهار', 'پنج', 'شش', 'هفت', 'هشت', 'نه', 'ده', 'یازده', 'دوازده'];
   222→const speakCache = new Map();
   223→let speakingBusy = false;
   224→
   225→// fallback محلی: Web Speech API خود مرورگر (fa-IR) — در استقرار مستقل (گیت‌هاب Pages) هم کار می‌کند؛
   226→// اگر صدای فارسی نصب نباشد، موتور پیش‌فرض تلاش می‌کند و در بدترین حالت بی‌صدا می‌ماند.
   227→function localSpeak(text) {
   228→  return new Promise(resolve => {
   229→    try {
   230→      if (typeof window === 'undefined' || !('speechSynthesis' in window)) return resolve(false);
   231→      const synth = window.speechSynthesis;
   232→      try { synth.cancel(); } catch { /* بی‌اهمیت */ }
   233→      const u = new SpeechSynthesisUtterance(text);
   234→      u.lang = 'fa-IR';
   235→      u.rate = 0.9;
   236→      const voices = typeof synth.getVoices === 'function' ? (synth.getVoices() || []) : [];
   237→      const faVoice = voices.find(v => v && /^fa/i.test(v.lang || ''));
   238→      if (faVoice) u.voice = faVoice;
   239→      let settled = false;
   240→      const done = ok => { if (!settled) { settled = true; resolve(ok); } };
   241→      u.onend = () => done(true);
   242→      u.onerror = () => done(false);
   243→      synth.speak(u);
   244→      setTimeout(() => done(true), 20000); // محافظ: اگر رویداد end/error نیامد
   245→    } catch { resolve(false); }
   246→  });
   247→}

   249→// گویندهٔ مشترک: صدا از API داشبورد (اگر در دسترس بود) → وگرنه صدای محلی مرورگر → وگرنه بی‌صدا.
   250→// فقط URLهای موفق کش می‌شوند تا شکست API در تلاش بعدی دوباره امتحان شود.
   251→async function speakFa(text, btn = null) {
   252→  if (!text || speakingBusy) return;
   253→  speakingBusy = true;
   254→  const prevLabel = btn ? btn.textContent : null;
   255→  if (btn) btn.textContent = '... در حال خواندن';
   256→  try {
   257→    let url = speakCache.get(text) || null;
   258→    if (!url && typeof fetch === 'function') {
   259→      try {
   260→        const res = await fetch('/api/tts', {
   261→          method: 'POST',
   262→          headers: { 'Content-Type': 'application/json' },
   263→          body: JSON.stringify({ text, speed: 0.9 })
   264→        });
   265→        if (res.ok) {
   266→          url = URL.createObjectURL(await res.blob());
   267→          speakCache.set(text, url);
   268→        }
   269→      } catch { /* داشبورد در دسترس نیست — fallback محلی */ }
   270→    }
   271→    if (url) {
   272→      await new Audio(url).play();
   273→    } else {
   274→      const ok = await localSpeak(text);
   275→      if (!ok && btn && btn.id === 'question-speak') notice('خواندن صوتی در دسترس نیست.', true);
   276→    }
   277→  } catch { /* TTS اختیاری است — بی‌صدا */ }
   278→  finally {
   279→    speakingBusy = false;
   280→    if (btn && prevLabel) btn.textContent = prevLabel;
   281→  }
   282→}

   734→async function handleAnswerSubmit(userAnswer) {
   735→  const session = state.session;
   736→  const question = session.questions[activeQuestionIndex];
   737→  const expected = question.f1 * question.f2;
   738→  const isCorrect = Number(normalizeDigits(userAnswer)) === expected;
   739→  // زمان‌سنجی سؤال (گرد تا یک دهم ثانیه) — برای receipt و تلاش‌نگار
   740→  const elapsedSec = questionStartedAt
   741→    ? Math.round(((Date.now() - questionStartedAt) / 1000) * 10) / 10
   742→    : 0;
   743→
   744→  const result = stationLayer.cells.onAnswer(isCorrect);
   745→
   746→  if (!isCorrect) {
   747→    // تلاش‌نگار فقط‌الحاقی (CEAP §4.1): پاسخ‌های غلط حذف نمی‌شوند — شاهد تشخیص گلوگاه
   748→    const submittedNum = Number(normalizeDigits(userAnswer));
   749→    // اطلس ۱۲خطا (v10.2): کد ساختاری از خود پاسخ + کد رفتاری از تاریخچهٔ پذیرش جلسه
   750→    const structuralCode = classifyAttempt({ f1: question.f1, f2: question.f2, submitted: submittedNum });
   751→    const behavioralCode = detectBehavioralCode({ f1: question.f1, f2: question.f2, acceptedFacts: session.acceptedFacts });
   752→    session.wrongAttempts = session.wrongAttempts || [];
   753→    session.wrongAttempts.push({
   754→      questionId: question.questionId,
   755→      f1: question.f1,
   756→      f2: question.f2,
   757→      submitted: submittedNum,
   758→      code: structuralCode || null,
   759→      behavioral: behavioralCode || null,
   760→      timeSec: elapsedSec,
   761→      at: new Date().toISOString()
   762→    });
   763→    $('question-feedback').textContent = result.feedback;

   943→function generateQuestionsForDepth(buildingId, depth) {
   944→  const layerKey = `${buildingId}-${depth}`;
   945→  const layer = DEPTH_LAYERS[layerKey] || DEPTH_LAYERS['1-1'];
   946→  const domain = layer.domain || 'table';
   947→  const base = (f1, f2, i, visual) => ({
   948→    questionId: `q_${buildingId}_${depth}_${i + 1}`,
   949→    f1,
   950→    f2,
   951→    ...(visual ? { visual } : {})
   952→  });
   953→
   954→  if (domain === 'x10') return X10_QUESTIONS.map((q, i) => base(q.f1, q.f2, i));
   955→  if (domain === 'zero-one') return ZERO_ONE_QUESTIONS.map((q, i) => base(q.f1, q.f2, i));
   956→  if (domain === 'multi-digit') return MULTI_DIGIT_QUESTIONS.map((q, i) => base(q.f1, q.f2, i));
   957→
   958→  if (domain === 'paired') {
   959→    const pairs = buildingId === 1 ? POWER_PAIRS : TOWER_PAIRS;
   960→    const out = [];
   961→    pairs.forEach(([a, b], k) => {
   962→      out.push(base(a, b, k * 2, 'equal-groups')); // سر اول زوج: تصویر دسته‌ها
   963→      out.push(base(b, a, k * 2 + 1));             // سر دوم: ترتیب چرخیده، بدون تصویر
   964→    });
   965→    return out;
   966→  }
   967→
   968→  // دامنهٔ جدول — لایهٔ ۱ نیروگاه کاملاً تصویری (CPA، فصل ۱.۲ سند)
   969→  const factors = layer.focus?.factors || [buildingId + 1];
   970→  const count = layer.questionCount || 8;
   971→  const visualAll = layerKey === '1-1';
   972→  const visualFirst = layerKey === '2-1';
   973→
   974→  const questions = [];
   975→  for (let i = 0; i < count; i++) {
   976→    const f1 = factors[i % factors.length];
   977→    const f2 = (i % 9) + 2;
   978→    const visual = (visualAll || (visualFirst && i < 4)) ? 'equal-groups' : null;
   979→    questions.push(base(f1, f2, i, visual));
   980→  }
   981→  return questions;
   982→}
