   165→def classify_attempt(f1, f2, submitted):
   166→    """پورت خط‌به‌خط classifyAttempt؛ خروجی: کد خطا یا None (درست/بی‌الگو)."""
   167→    if not all(isinstance(x, int) for x in (f1, f2, submitted)):
   168→        return None
   169→    if f1 < 0 or f2 < 0 or f1 > 999 or f2 > 999:
   170→        return None
   171→    if submitted < 0 or submitted > 9999:
   172→        return None
   173→
   174→    p = f1 * f2
   175→    if submitted == p:
   176→        return None  # پاسخ درست
   177→
   178→    # ۱. ضرب در صفر
   179→    if f1 == 0 or f2 == 0:
   180→        other = max(f1, f2)
   181→        if submitted == other or submitted == 1:
   182→            return 'E_ZERO_PROPERTY'
   183→        return None
   184→
   185→    # ۲. ضرب در یک
   186→    if (f1 == 1 or f2 == 1) and submitted == 1:
   187→        return 'E_IDENTITY_CONFUSION'
   188→
   189→    # ۳. جمع به جای ضرب
   190→    if f1 >= 2 and f2 >= 2 and submitted == f1 + f2:
   191→        return 'E_ADD_CONFUSION'
   192→
   193→    # ۴. لغزش صفر (ارزش مکانی)
   194→    if p > 0 and (submitted == p * 10 or (p % 10 == 0 and submitted == p // 10)):
   195→        return 'E_PLACE_SHIFT'
   196→
   197→    # ۵. جا ماندن جزء دوم شکستن (لنگر ۵)
   198→    if f2 > 5 and submitted == f1 * 5:
   199→        return 'E_ANCHOR_DRIFT'
   200→    if f1 > 5 and submitted == f2 * 5:
   201→        return 'E_ANCHOR_DRIFT'
   202→
   203→    # ۶. جا ماندن یک جزء چندرقمی
   204→    if f2 >= 11 and 2 <= f1 <= 9:
   205→        tens, ones = divmod(f2, 10)
   206→        if ones > 0 and submitted == f1 * ones:
   207→            return 'E_DISTRIBUTIVE_LEAK'
   208→        if tens > 0 and submitted in (f1 * tens, f1 * tens * 10):
   209→            return 'E_DISTRIBUTIVE_LEAK'
   210→    if f1 >= 11 and 2 <= f2 <= 9:
   211→        tens, ones = divmod(f1, 10)
   212→        if ones > 0 and submitted == f2 * ones:
   213→            return 'E_DISTRIBUTIVE_LEAK'
   214→        if tens > 0 and submitted in (f2 * tens, f2 * tens * 10):
   215→            return 'E_DISTRIBUTIVE_LEAK'
   216→
   217→    # ۷ و ۸. بیش/کم‌شماری جهش
   218→    if f2 >= 2 and submitted == p + f2:
   219→        return 'E_SKIP_OVERCOUNT'
   220→    if f2 >= 2 and submitted == p - f2:
   221→        return 'E_SKIP_UNDERCOUNT'
   222→    if f1 >= 2 and submitted == p + f1:
   223→        return 'E_SKIP_OVERCOUNT'
   224→    if f1 >= 2 and submitted == p - f1:
   225→        return 'E_SKIP_UNDERCOUNT'
   226→
   227→    return None
   228→

   234→def parse_option_value(raw):
   235→    """مقدار عددی گزینه: عدد خام / جمع / ضرب / «همان N» — وگرنه None."""
   236→    if raw is None:
   237→        return None
   238→    text = str(raw).strip()
   239→    t = re.sub(r'\s+', '', to_latin(text))
   240→    if re.fullmatch(r'-?\d+', t):
   241→        return int(t)
   242→    if '+' in t and '×' not in t and all(re.fullmatch(r'\d+', p) for p in t.split('+')):
   243→        return sum(int(p) for p in t.split('+'))
   244→    if '×' in t and '+' not in t and all(re.fullmatch(r'\d+', p) for p in t.split('×')):
   245→        prod = 1
   246→        for p in t.split('×'):
   247→            prod *= int(p)
   248→        return prod
   249→    m = re.search(r'همان\s*([۰-۹0-9]+)', text)
   250→    return int(to_latin(m.group(1))) if m else None

   253→def parse_plain_multiplication(prompt):
   254→    """فرم ضربِ خام «a×b = ؟» → {f1, f2}؛ وگرنه None (فقط این‌ها بازحساب مستقیم می‌شوند)."""
   255→    m = re.match(r'^\s*([0-9۰-۹]+)\s*[×x]\s*([0-9۰-۹]+)\s*=\s*؟\s*$', str(prompt or '').strip())
   256→    if not m:
   257→        return None
   258→    return {'f1': int(to_latin(m.group(1))), 'f2': int(to_latin(m.group(2)))}

   261→def extract_multiplication_factors(prompt):
   262→    """عامل‌های ضرب درون متن سؤال (حتی غیرخام)؛ ÷ یا ☐ → None."""
   263→    text = str(prompt or '')
   264→    if '÷' in text or '☐' in text:
   265→        return None
   266→    m = re.search(r'([0-9۰-۹]+)\s*[×x]\s*([0-9۰-۹]+)', text)
   267→    if not m:
   268→        return None
   269→    return {'f1': int(to_latin(m.group(1))), 'f2': int(to_latin(m.group(2)))}

   272→def parse_claims(map_text):
   273→    """«الف→…؛ ب→…» → [{letter, claim}] — با همان برشِ ؛ قرارداد schema."""
   274→    out = []
   275→    text = str(map_text or '')
   276→    for letter in OPTION_LETTERS:
   277→        m = re.search(re.escape(letter) + r'→([^؛]+)', text)
   278→        if m:
   279→            out.append({'letter': letter, 'claim': m.group(1).strip()})
   280→    return out

   283→def extract_q_refs(text):
   284→    """ارجاع‌های Q- درون سلول، با گسترش «Q-۱۰۳ تا Q-۱۰۵»."""
   285→    t = str(text or '')
   286→    refs, consumed = [], []
   287→    for m in re.finditer(r'Q-([0-9۰-۹]+)\s*تا\s*Q-([0-9۰-۹]+)', t):
   288→        a, b = int(to_latin(m.group(1))), int(to_latin(m.group(2)))
   289→        if b >= a and b - a < 50:
   290→            refs.extend(f'Q-{i}' for i in range(a, b + 1))
   291→        consumed.append(m.group(0))
   292→    rest = t
   293→    for c in consumed:
   294→        rest = rest.replace(c, ' ')
   295→    refs.extend(re.findall(r'Q-[0-9۰-۹]+', rest))
   296→    return refs

   340→def header_map(ws, anchor_label, max_scan=10):
   341→    """نقشه «سرستون نرمال‌شده → اندیس ستون» برای اولین ردیفی که ستونِ لنگر آن = anchor."""
   342→    for r in range(1, min(ws.max_row, max_scan) + 1):
   343→        cols = {}
   344→        for c in range(1, ws.max_column + 1):
   345→            h = norm(ws.cell(row=r, column=c).value)
   346→            if h:
   347→                cols.setdefault(h, c)
   348→        if anchor_label in cols:
   349→            return r, cols
   350→    return None, {}

   354→    """ردیف‌های داده: از بعد هدر تا ردیف پانویس/تهی. پانویس = «یادداشت»/«قاعدهٔ»/«نتیجه»."""
   355→    for r in range(header_row + 1, ws.max_row + 1):

   357→        any_val = any(cell_text(ws.cell(row=r, column=c).value) not in (None, '')

   362→                                or idv.strip().startswith(('قاعدهٔ', 'نتیجه'))):

   570→def enum_from_glossary(glossary, column_label, fallback):
   571→    """مقادیر مجاز یک ستون از شیت واژه‌نامه («الف | ب | ج»)؛ نبودِ ردیف → fallback."""
   572→    for g in glossary:

   576→                return vals
   577→    return list(fallback)

   587→def judge_claim(qid, letter, claim, f1, f2, submitted, atlas_proposed):
   588→    """قضاوت یک ادعای گزینه با موتور واقعی — عین منطق ممیزی AUD-01."""
   589→    if (qid, letter) in KNOWN_HALF_MAPS:
   590→        return ('warning', f'نقشهٔ نیمه‌تزئینی — زرد مستند AUD-12 ({qid} گ{letter})')
   591→    has_proposed_marker = re.search(r'پیشنهادی|کاندید', claim) is not None
   592→    code = claim_code(claim)
   593→    engine = classify_attempt(f1, f2, submitted)
   594→
   595→    if code and code not in REAL_CODES:
   596→        if code in atlas_proposed:
   597→            return None  # کد پیشنهادیِ ثبت‌شده در اطلس — تصمیم با S2
   598→        return ('error', f'کد بیرونیِ بدون تگ در اطلس: {code} (T5 — قرمز)')
   599→    if code and code in REAL_CODES:
   600→        if re.search(r'هم‌پوشانی', claim):
   601→            if engine in REAL_CODES:
   602→                return None
   603→            return ('error', f'ادعای هم‌پوشانی ولی موتور بی‌الگو داد ({engine})')
   604→        if engine == code:
   605→            return None
   606→        return ('error', f'ادعا {code} ولی موتور {engine if engine else "بی‌الگو"} داد — ناسازگاری T6')
   607→    if re.search(r'بی‌الگو|چسباندن', claim):
   608→        if engine is None:
   609→            return None
   610→        return ('error', f'ادعای بی‌الگو ولی موتور {engine} زد — ناسازگاری T6')
   611→    return None  # توصیف متنی بدون کد — خارج از نگاشت خودکار

   614→def validate(cartridge, enums, red_rows):
   615→    """خروجی: {ok, errors, warnings, stats, gate} — fail-closed."""
   616→    errors, warnings = [], []
   617→    err, warn = errors.append, warnings.append
   618→
   619→    bookmap = cartridge['bookMap']
   620→    atlas = cartridge['errorAtlas']
   621→    bank = cartridge['questionBank']
   622→    cpa_values, engine_values, category_values = enums
   623→
   624→    # قرمزهای مبدأ (ردیف‌های «ایراد» که عمداً به JSON نمی‌روند — قاعدهٔ رنگ)
   625→    for sheet, label in red_rows:
   626→        err(f'ردیف قرمز (ایراد مستند) در شیت «{sheet}»: {label} — به JSON نمی‌رود (قاعدهٔ رنگ)')
   627→
   628→    # ── اطلس خطا (T5 + رده AUD-08) ──
   629→    atlas_proposed, atlas_codes = set(), set()
   630→    for a in atlas:
   631→        code = a.get('code')
   632→        if not code:
   633→            err('ردیف اطلس بدون کد')
   634→            continue
   635→        if code in atlas_codes:
   636→            err(f'کد تکراری در اطلس: {code}')
   637→        atlas_codes.add(code)
   638→        if a['status'] == STATUS_PROPOSED:
   639→            atlas_proposed.add(code)
   640→        elif code not in REAL_CODES:
   641→            err(f'اطلس: {code} «تأییدشده» است ولی در errors.js نیست (T5)')
   642→        if a['status'] == STATUS_PROPOSED and code in REAL_CODES:
   643→            warn(f'اطلس: {code} پیشنهادی است ولی هم‌نام کد واقعی موتور است')
   644→        cat = a.get('category')
   645→        if cat:
   646→            # توکِل‌بندی محتاطانه: ارزش مرکب مستند (مثل «مفهومی/محاسباتی») می‌گذرد؛
   647→            # هر توکِل ناشناخته = زرد (AUD-08 — توصیف اضافه باید در یادداشت باشد)
   648→            tokens = [t for t in re.split(r'[/\s]+', str(cat).strip()) if t]
   649→            if not all(t in category_values or t in CATEGORY_EXTRA_TOKENS for t in tokens):
   650→                warn(f'اطلس خطا {code}: مقدار «رده» خارج از واژه‌نامه: «{cat}» (زرد AUD-08)')
   651→    for a in atlas:
   652→        for ref in extract_q_refs(a.get('questionPattern')):
   653→            if not any(q['id'] == ref for q in bank):
   654→                err(f'اطلس {a["code"]}: ارجاع {ref} در بانک سؤال نیست')
   655→
   656→    # ── نقشهٔ کتاب (قاعدهٔ ۸ + سازگاری موتور AUD-07) ──
   657→    lesson_ids = set()
   658→    for l in bookmap:
   659→        lid = l.get('id')
   660→        if not lid:
   661→            err('ردیف نقشهٔ کتاب بدون شناسه')
   662→            continue
   663→        if lid in lesson_ids:
   664→            err(f'شناسهٔ تکراری درس: {lid}')
   665→        lesson_ids.add(lid)
   666→        if l['status'] == STATUS_CONFIRMED and not (l.get('source') or '').strip():
   667→            err(f'نقشهٔ کتاب {lid}: ردیف سبز بدون منبع (قاعدهٔ ۸)')
   668→        ec = l.get('engineCompat')
   669→        if ec and ec not in engine_values:
   670→            warn(f'نقشهٔ کتاب {lid}: مقدار «سازگاری موتور» خارج از واژه‌نامه: «{ec}» (زرد AUD-07)')
   671→
   672→    # ── بانک سؤال (T6 + ضد J12 + AUD-06/AUD-12) ──
   673→    question_ids = set()
   674→    derived_owners = {}  # lessonId → [Q-…]
   675→    for q in bank:
   676→        qid = q.get('id')
   677→        if not qid:
   678→            err('ردیف بانک بدون شناسه')
   679→            continue
   680→        if qid in question_ids:
   681→            err(f'شناسهٔ تکراری سؤال: {qid}')
   682→        question_ids.add(qid)
   683→        if q.get('anchorId') not in lesson_ids:
   684→            err(f'{qid}: لنگر «{q.get("anchorId")}» در نقشهٔ کتاب نیست')
   685→        derived_owners.setdefault(q.get('anchorId'), []).append(qid)
   686→
   687→        if q['status'] == STATUS_CONFIRMED and not (q.get('source') or '').strip():
   688→            err(f'{qid}: ردیف سبز بدون منبع (قاعدهٔ ۸)')
   689→        if q.get('domain') and q['domain'] not in ('CORE', 'DEPTH', 'SIDE'):
   690→            warn(f'{qid}: دامنهٔ خارج از واژه‌نامه: {q["domain"]}')
   691→        if q.get('cpa') and q['cpa'] not in cpa_values:
   692→            warn(f'{qid}: لایهٔ CPA خارج از واژه‌نامه: «{q["cpa"]}» (زرد AUD-06)')
   693→
   694→        options = q.get('options') or []
   695→        if len(options) != 4:
   696→            err(f'{qid}: تعداد گزینه‌ها ≠ ۴')
   697→            continue
   698→        ai = q.get('answerIndex')
   699→        if not isinstance(ai, int) or ai < 0 or ai > 3:
   700→            err(f'{qid}: پاسخ‌نام خارج از بازه')
   701→        for i, opt in enumerate(options):
   702→            v = parse_option_value(opt)
   703→            if v is not None and (v < 0 or v > 9999):
   704→                err(f'{qid} گ{OPTION_LETTERS[i]}: عدد خارج از دامنهٔ کتاب (۹۹۹۹ — T4)')
   705→
   706→        # کلید پاسخ: ابتدا فرم خام (a×b=؟) با بازحساب — ضد J12؛ سپس رسید ممیزی
   707→        plain = parse_plain_multiplication(q.get('prompt'))
   708→        if plain and isinstance(ai, int) and 0 <= ai <= 3:
   709→            v = parse_option_value(options[ai])
   710→            if v != plain['f1'] * plain['f2']:
   711→                err(f'{qid}: گزینهٔ پاسخ ({v}) ≠ حاصل {plain["f1"]}×{plain["f2"]}={plain["f1"] * plain["f2"]} (ضد J12)')
   712→        elif qid in AUDIT_ANSWER_RECEIPTS and isinstance(ai, int) and 0 <= ai <= 3:
   713→            v = parse_option_value(options[ai])
   714→            if v != AUDIT_ANSWER_RECEIPTS[qid]:
   715→                err(f'{qid}: گزینهٔ پاسخ ({v}) ≠ رسید ممیزی ({AUDIT_ANSWER_RECEIPTS[qid]})')
   716→
   717→        # برچسب گزینه→خطا با موتور واقعی
   718→        factors = extract_multiplication_factors(q.get('prompt'))
   719→        if factors:
   720→            claims = q.get('claims') or []
   721→            for i, letter in enumerate(OPTION_LETTERS):
   722→                claim_obj = next((c for c in claims if c['letter'] == letter), None)
   723→                if not claim_obj:
   724→                    continue
   725→                v = parse_option_value(options[i])
   726→                if v is None or v == factors['f1'] * factors['f2']:
   727→                    continue
   728→                verdict = judge_claim(qid, letter, claim_obj['claim'],
   729→                                      factors['f1'], factors['f2'], v, atlas_proposed)
   730→                if verdict and verdict[0] == 'error':
   731→                    err(f'{qid} گ{letter}={v}: {verdict[1]}')
   732→                if verdict and verdict[0] == 'warning':
   733→                    warn(f'{qid} گ{letter}: {verdict[1]}')
   734→
   735→    # ── سازگاری نقشه ↔ بانک (T7 — بستن AUD-04/AUD-05) ──
   736→    for l in bookmap:
   737→        lid = l.get('id')
   738→        if not lid:
   739→            continue
   740→        refs = sorted(set(extract_q_refs(l.get('anchorsRaw'))))
   741→        derived = sorted(derived_owners.get(lid) or [])
   742→        if refs != derived:
   743→            err(f'نقشهٔ کتاب {lid}: لنگرهای سلول [{ "،".join(refs) }] ≠ مالکان واقعی در بانک [{ "،".join(derived) }]')
   744→        for ref in refs:
   745→            if ref not in question_ids:
   746→                err(f'نقشهٔ کتاب {lid}: ارجاع {ref} در بانک سؤال نیست')
   747→
   748→    for note in MISMATCH_NOTES:
   749→        warn(f'ناهم‌خوانی رنگ/متن «رنگ اعتبار» — {note}')
   750→
   751→    stats = {
   752→        'lessons': len(bookmap),
   753→        'lessonsConfirmed': sum(1 for l in bookmap if l['status'] == STATUS_CONFIRMED),
   754→        'atlasReal': sum(1 for a in atlas if a['status'] == STATUS_CONFIRMED),
   755→        'atlasProposed': len(atlas_proposed),
   756→        'questions': len(bank),
   757→        'questionsConfirmed': sum(1 for q in bank if q['status'] == STATUS_CONFIRMED),
   758→        'questionsProposed': sum(1 for q in bank if q['status'] == STATUS_PROPOSED),
   759→    }
   760→    ok = len(errors) == 0
   761→    # بلوک validation داخل JSON همان قرارداد schema است (ok/warnings/stats/gate —
   762→    # بدون آرایهٔ errors)؛ فهرست کامل خطاها فقط به گزارش md می‌رود.
   763→    validation = {
   764→        'ok': ok,
   765→        'warnings': warnings,
   766→        'stats': stats,
   767→        'gate': {'red': len(errors), 'yellow': len(warnings),
   768→                 'verdict': 'مجاز' if ok else 'ممنوع'},
   769→    }
   770→    return validation, errors
   771→

   777→def render_report(cartridge_id, input_name, when, validation, errors, published):
   778→    v = validation
   779→    s = v['stats']
   780→    lines = []
   781→    lines.append(f'# گزارش مبدل خشاب — {cartridge_id}')
   782→    lines.append('')
   783→    lines.append(f'- ورودی: `{input_name}`')
   784→    lines.append(f'- ابزار: {TOOL_NAME} (قواعد = پرامپت ممیز T1..T7 + رسیدهای ممیزی)')
   785→    lines.append(f'- زمان: {when}')
   786→    lines.append('')
   787→    lines.append('| سنجه | مقدار |')
   788→    lines.append('|---|---|')
   789→    red, yellow = v['gate']['red'], v['gate']['yellow']
   790→    lines.append(f'| قرمز (بلوکه‌کننده) | **{red}** |')
   791→    lines.append(f'| زرد (هشدار) | {yellow} |')
   792→    lines.append(f'| درس‌ها | {s["lessons"]} (سبز: {s["lessonsConfirmed"]}) |')
   793→    lines.append(f'| اطلس | {s["atlasReal"]} واقعی + {s["atlasProposed"]} پیشنهادی |')
   794→    lines.append(f'| بانک سؤال | {s["questions"]} (سبز: {s["questionsConfirmed"]}، زرد: {s["questionsProposed"]}) |')
   795→    lines.append('')
   796→    lines.append('## خطاها')
   797→    if not errors:
   798→        lines.append('- هیچ — گیت سبز.')
   799→    else:
   800→        for e in errors:
   801→            lines.append(f'- 🔴 {e}')
   802→    lines.append('')
   803→    lines.append('## هشدارها (زرد — همان‌هایی که ممیزی AUD رسماً ثبت کرده)')
   804→    for w in v['warnings']:
   805→        lines.append(f'- 🟡 {w}')
   806→    lines.append('')
   807→    lines.append('## رسیدهای کلید پاسخ')
   808→    lines.append(f'بررسی خودکار علیه {len(AUDIT_ANSWER_RECEIPTS)} رسید ممیزی + بازحساب فرم خام a×b.')
   809→    lines.append('')
   810→    if published:
   811→        lines.append(f'**حکم گیت مبدل: مجاز — JSON منتشر شد**')
   812→    else:
   813→        lines.append(f'**حکم گیت مبدل: ممنوع — JSON نوشته نشد (fail-closed)**')
   814→    return '\n'.join(lines)
