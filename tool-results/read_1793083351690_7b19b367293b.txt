   354→   354→async function flush(force = false) {
   355→   355→  if (!actor || actor.teacher) return;
   356→   356→  if (!navigator.onLine) { notice('اینترنت قطع است؛ پاسخ‌ها روی این دستگاه نگهداری می‌شوند.'); return; }
   357→   357→  const commands = await local.pending(actor.uid);
   358→   358→  for (const command of commands) {
   359→   359→    if (command.blocked) { notice('بعضی پاسخ‌ها نیاز به بررسی معلم دارند: ' + command.blocked, true); continue; }
   360→   360→    if (!force && command.retryAt > Date.now()) break;
   361→   361→    try {
   362→   362→      const result = await api('submitAttempts', command.input);
   363→   363→      await local.acknowledge(command, result, key());
   364→   364→      const last = result.receipts.at(-1);
   365→   365→      if (state.session?.sessionId === result.sessionId && (!command.stateKey || command.stateKey === key())) {
   366→   366→        const attempt = command.input.attempts.find(attempt => attempt.eventId === last.eventId);
   367→   367→        const question = state.session.questions.find(question => question.questionId === attempt?.questionId);
   368→   368→        $('feedback').textContent = last.correct ? 'پاسخ قبلی درست بود و در پرونده تأیید شد؛ آفرین!' : 'پاسخ قبلی نیاز به مرور داشت؛ جواب درست ' + fa(question.f1 * question.f2) + ' بود. پاسخ تو در پرونده تأیید شد.';
   369→   369→      }
   370→   370→    } catch (error) {
   371→   371→      if (['FORBIDDEN', 'EVENT_CONFLICT', 'STALE_WRITER', 'INVALID_PAYLOAD'].includes(error.code)) await local.block(command, error.code);
   372→   372→      else await local.retry(command);
   373→   373→      notice(error.code ? error.message : 'پاسخ روی این دستگاه ذخیره است؛ ارسال با برگشت ارتباط ادامه می‌یابد.', true);
   374→   374→      break;
   375→   375→    }
   376→   376→  }
   377→   377→  state = await local.get(key()) || {}; render();
   378→   378→}
